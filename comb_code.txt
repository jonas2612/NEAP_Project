from pathlib import Path

base_path = Path("/usr/local/storage/data_microarray")

tmp = pd.read_csv("/usr/local/storage/data_microarray/260927_normalized_miRNA/comb_GSE137580_GSE205119_norm_data.tsv", sep="\t", index_col=0)

data_path_padj = out_D_view / "GSE205119_GSE137580_hsa_circ_0122319_-_WT_padj.tsv"
data_path_logfc = out_D_view / "GSE205119_GSE137580_hsa_circ_0122319_-_WT_logfc.tsv"

dea = pd.read_csv("/usr/local/storage/data_microarray/260927_dea_miRNA/GSE205119_GSE137580_hsa_circ_0122319_-_WT_dea.tsv", sep="\t")
dea['-log10(p.adj)'] = -np.log10(dea['adj.P.Val'].values)
dea_pval = pd.DataFrame(
            dea["-log10(p.adj)"].values, 
            columns=["-log10(p.adj)"], 
            index = dea["GeneName"].values
            )
dea_logfc = pd.DataFrame(
            dea["logFC"].values,
            columns=["logFC"], 
            index = dea["GeneName"].values
            )
dea_pval.to_csv(data_path_padj, sep="\t")
dea_logfc.to_csv(data_path_logfc, sep="\t")
